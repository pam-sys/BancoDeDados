CREATE VIEW cinco_produtos_mais_vendidos AS
SELECT
    p.produtoid,
    p.nome AS produto,
    SUM(iv.quantidade) AS quantidade_total_vendida
FROM produto p
JOIN itemvenda iv
    ON p.produtoid = iv.produtoid
GROUP BY p.produtoid, p.nome
ORDER BY quantidade_total_vendida DESC
LIMIT 5;
